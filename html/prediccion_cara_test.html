import numpy as np
from sklearn.metrics import log_loss

# Número de observaciones y parámetros
n = X_train.shape[0]
k = np.count_nonzero(best_model.coef_) + 1  # solo coeficientes ≠ 0 + intercepto
k_best = np.count_nonzero(best_model.coef_) + 1
k_default = np.count_nonzero(model.coef_) + 1

# Calcular log-verosimilitud negativa
log_likelihood_best = -log_loss(y_train, best_model.predict_proba(X_train), normalize=False)
log_likelihood_model = -log_loss(y_train, model.predict_proba(X_train), normalize=False)

# Calcular AIC y BIC
# AIC y BIC
aic_best = 2 * k_best - 2 * log_likelihood_best
bic_best = k_best * np.log(n) - 2 * log_likelihood_best
aic_default = 2 * k_default - 2 * log_likelihood_model
bic_default = k_default * np.log(n) - 2 * log_likelihood_model

import time
import matplotlib.pyplot as plt
import seaborn as sns
import numpy as np
import pandas as pd
from sklearn.tree import DecisionTreeClassifier
from sklearn.ensemble import RandomForestClassifier
from sklearn.naive_bayes import GaussianNB
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import (
    confusion_matrix, ConfusionMatrixDisplay,
    classification_report, roc_curve, auc
)
from sklearn.model_selection import train_test_split

# Separar X e y (usando los datos ya preprocesados)
X = train_data.drop(columns=["EsCara"])
y = train_data["EsCara"]

# División para evaluación consistente
X_train_split, X_test_split, y_train_split, y_test_split = train_test_split(X, y, test_size=0.3, random_state=42)

# Diccionario para almacenar resultados
modelos = {
    "Regresión Logística": LogisticRegression(C=0.1, penalty="l1", solver="liblinear", max_iter=1000),
    "Árbol de Decisión": DecisionTreeClassifier(random_state=42),
    "Random Forest": RandomForestClassifier(n_estimators=100, random_state=42),
    "Naive Bayes": GaussianNB()
}

resultados = {}

# Entrenar y evaluar cada modelo
for nombre, modelo in modelos.items():
    inicio = time.time()
    modelo.fit(X_train_split, y_train_split)
    fin = time.time()
    
    y_pred = modelo.predict(X_test_split)
    y_proba = modelo.predict_proba(X_test_split)[:, 1]
    
    cm = confusion_matrix(y_test_split, y_pred)
    reporte = classification_report(y_test_split, y_pred, output_dict=True)
    fpr, tpr, _ = roc_curve(y_test_split, y_proba)
    roc_auc = auc(fpr, tpr)
    
    resultados[nombre] = {
        "modelo": modelo,
        "tiempo": round(fin - inicio, 4),
        "confusion_matrix": cm,
        "reporte": reporte,
        "roc_curve": (fpr, tpr),
        "roc_auc": roc_auc
    }

# === GRAFICAR MATRICES DE CONFUSIÓN ===
fig, axes = plt.subplots(2, 2, figsize=(12, 10))
axes = axes.ravel()

for i, (nombre, datos) in enumerate(resultados.items()):
    disp = ConfusionMatrixDisplay(confusion_matrix=datos["confusion_matrix"])
    disp.plot(ax=axes[i], cmap="Blues")
    axes[i].set_title(f"Matriz de Confusión: {nombre}")
plt.tight_layout()
plt.show()

# === GRAFICAR CURVAS ROC ===
plt.figure(figsize=(8, 6))
for nombre, datos in resultados.items():
    fpr, tpr = datos["roc_curve"]
    roc_auc = datos["roc_auc"]
    plt.plot(fpr, tpr, label=f"{nombre} (AUC = {roc_auc:.2f})")
plt.plot([0, 1], [0, 1], "k--")
plt.xlabel("FPR")
plt.ylabel("TPR")
plt.title("Curvas ROC")
plt.legend()
plt.grid(True)
plt.tight_layout()
plt.show()

# === MOSTRAR MÉTRICAS DE COMPARACIÓN ===
for nombre, datos in resultados.items():
    print(f"\n {nombre}")
